\begin{frame}[fragile]{Step 10: Step 8 as a function}
    \begin{columns}[T]
        \begin{column}{0.37\textwidth}

            \medskip
            So put the request in a function that takes the page.
        \end{column}
        \begin{column}{0.6\textwidth}
\begin{lstlisting}
def extract_totals(page):
    body = {"models": MODELS, "temperature": 0,
            "messages": [..., {"role": "user",
                               "content": page}], ...}
    response = requests.post(
        f"{openrouter_base}/chat/completions",
        json=body, headers=...).json()
    content = response["choices"][0]["message"]["content"]
    return json.loads(content), response["model"]
\end{lstlisting}
        \end{column}
    \end{columns}

\begin{frame}[fragile]{Step 13: search by meaning}
    \begin{columns}[T]
        \begin{column}{0.37\textwidth}
            Embed the question as Step 12 embedded a lead. Score every lead against it, and keep that article's best three.

def search(question, title):
    response = requests.post(
        f"{openrouter_base}/embeddings", headers=...,
        json={"model": EMBEDDER, "input": [question]}).json()
    query = np.array(response["data"][0]["embedding"])
    query = query / np.linalg.norm(query)
    leads["score"] = vectors @ query
    best = leads[leads["title"] == title]
    return best.nlargest(3, "score")

\begin{frame}{Key takeaways}
    \begin{enumerate}
        \item Call a model \textbf{as a function}: one request in, JSON or one tool call out. Your code decides what runs.
        \item \textbf{Valid JSON can still be wrong.} Check it against something you trust.
        \item Datalab \textbf{read} some of its output and its model \textbf{wrote} the rest.
        \item The same request can return a different answer. \textbf{Record} the model, the date, and the response.
        \item A similarity says how much a text changed. \textbf{Read} it to learn why.
    \end{enumerate}
