\begin{frame}[fragile]{Step 3: a function that splits pages}
    \begin{columns}[T]
        \begin{column}{0.37\textwidth}
            Each page starts with a marker such as \texttt{\{14\}} and a line of dashes. This given function splits on the markers.

            \medskip
            The pattern is a \textbf{regular expression}: a short way to describe text.
        \end{column}
        \begin{column}{0.6\textwidth}
\begin{lstlisting}
# Split at each page marker; drop the empty first piece
def split_pages(markdown):
    pieces = re.split(
        r"\n*\{\d+\}-{10,}\n*",
        markdown,
    )
    return pieces[1:]
\end{lstlisting}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{What is a text embedding?}

    \begin{columns}
        \begin{column}{0.6\textwidth}
            A word, sentence, or document embedding is a machine learning method for representing text as a numerical vector

            \medskip

            The vector is typically hundreds or thousands of elements long and the individual values or positions are not interpretable

            \medskip

            The ``position'' of the text in this high-dimensional space is inferred from text patterns observed in other training data

            \medskip

            The position encodes \textit{semantic} information: text that is close to other text in an embedding is more similar to each other

            \medskip

            These embeddings are not neutral and encode interesting and troubling biases from training data
        \end{column}
        \begin{column}{0.35\textwidth}\centering
            \includegraphics[width=\textwidth]{slides/week-13/img/embedding-gender-biases.png}\\ \footnotesize
            $x$-axis is male-female bias and $y$-axis is gender neutrality. From: Boulukbasi, et al. (2016)
        \end{column}
    \end{columns}

\begin{frame}{Key takeaways}
    \begin{enumerate}
        \item Call a model \textbf{as a function}: one request in, JSON or one tool call out. Your code decides what runs.
        \item \textbf{Valid JSON can still be wrong.} Check it against something you trust.
        \item Datalab \textbf{read} some of its output and its model \textbf{wrote} the rest.
        \item The same request can return a different answer. \textbf{Record} the model, the date, and the response.
        \item A similarity says how much a text changed. \textbf{Read} it to learn why.
    \end{enumerate}
\end{frame}
