\section{Monday · Concepts}
% =========================================================================

\begin{frame}{The golden age of web data access}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            For roughly a decade --- about \textbf{2008 to 2018} --- the web behaved as if it wanted to be studied.
            \begin{itemize}\small
                \item Twitter opened its API in 2006; Facebook launched the Graph API in 2010; Reddit's API was free and essentially unrestricted; Google exposed trends, maps, and books.
                \item This was not charity: platforms traded \textbf{data access} for \textbf{ecosystems}, credibility, and lock-in.
                \item The bargain built a field --- computational social science \parencite{lazer2009computational} --- and a generation of data journalism and civic tech.
            \end{itemize}

            \vspace{0.5em}

            When researchers today say the \textbf{``post-API age''} \parencite{freelon2018computational}, this is the era they measure against.
        \end{column}
        \begin{column}{0.36\textwidth}
            \begin{block}{Built on the bargain}
                \footnotesize The Wikipedia and Reddit studies that trained a generation --- including the instructor's --- ran on open APIs, public dumps, and volunteer archives like Pushshift \parencite{baumgartner2020pushshift}. No permission, partnership, or payment required.
            \end{block}
            {\scriptsize You will use those same open interfaces in Ch.~10 (Wikipedia).}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~3, ``The Golden Age of Web Data Access.''}
\end{frame}

\begin{frame}{Pressure 1 --- Enclosure}
    \begin{columns}[T]
        \begin{column}{0.56\textwidth}
            Three pressures explain the shift since $\sim$2018: \textbf{\textcolor{tab-blue}{enclosure}}, \textbf{\textcolor{tab-orange}{exemption}}, \textbf{\textcolor{tab-red}{erosion}}.

            \vspace{0.5em}

            \textbf{Enclosure} privatizes a shared resource --- the name echoes the fencing of English common land. The posts still exist; the \textit{terms of access} are fenced.
            \begin{itemize}\small
                \item \textbf{Twitter/X}: a free academic archive (2021) $\rightarrow$ discontinued; enterprise access from \$42{,}000/month. Hundreds of tools went dark.
                \item \textbf{Reddit}: free since 2008 $\rightarrow$ \$0.24 per 1{,}000 calls (2023); Apollo shut down; 7{,}000+ subreddits went dark; Pushshift was cut off.
            \end{itemize}
            \textbf{The driver: generative AI.} Once text became training data, an open API stopped looking like ecosystem-building and started looking like giving away inventory (Reddit--Google: $\sim$\$60M/year).
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/three_pressures.png}\\
            {\scriptsize Three forces fencing the open web.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~3, ``Enclosure.'' Facebook's CrowdTangle shutdown (2024) was the template.}
\end{frame}

\begin{frame}{Pressure 2 --- Exemption}
    \begin{columns}[T]
        \begin{column}{0.56\textwidth}
            \textbf{Exemption}: platforms operate beyond outside scrutiny while running the same studies internally at will.

            \vspace{0.5em}

            \textbf{The asymmetry is the tell.} Every A/B test is a behavioral study, and internal teams have total access --- but outside researchers meet:
            \begin{itemize}\small
                \item cease-and-desist letters citing the Computer Fraud and Abuse Act;
                \item Terms of Service that prohibit the very collection research needs \parencite{fiesler2020no};
                \item account bans --- e.g., Meta disabling NYU's Ad Observatory (2021), justified as \textit{protecting privacy}.
            \end{itemize}
            When ``privacy'' restricts only accountability research --- never ad targeting on the same users --- it is functioning as exemption, not protection.
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/ad_observatory.png}\\
            {\scriptsize Consenting users' data, still shut down.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~3, ``Exemption.'' Courts push back at the margins (\emph{Sandvig}, \emph{Van Buren}, \emph{hiQ}; see Ch.~2). EU DSA Article~40 may mandate vetted-researcher access.}
\end{frame}

\begin{frame}{Pressure 3 --- Erosion}
    \begin{columns}[T]
        \begin{column}{0.56\textwidth}
            \textbf{Erosion} is the quietest pressure: enclosure and exemption are things platforms \textit{do}; erosion is what happens to the web \textit{itself}.
            \begin{itemize}\small
                \item \textbf{Links rot.} About half the links in U.S. Supreme Court opinions are dead; a 2024 Pew study found about a third of 2013 web pages gone a decade later.
                \item \textbf{Platforms churn.} GeoCities vanished; Parler went offline overnight; Twitter became X and broke a decade of embedded links.
                \item \textbf{Sites redesign} and silently break every scraper --- fragility you will feel in Ch.~6 \& 8.
            \end{itemize}

            \vspace{0.3em}

            For research this is a \textbf{reproducibility crisis}: a study on the Twitter Academic API cannot be rerun at any price \parencite{davidson2023platform}.
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/link_rot.png}\\
            {\scriptsize The record decays unless someone saves it.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~3, ``Erosion.'' Two later chapters are counter-erosion: archives (Ch.~7) and automation (Ch.~14).}
\end{frame}

\begin{frame}{This is an old problem --- public-interest lineages}
    Data science is not the first profession to face powerful actors gatekeeping information the public needs. Each precedent offers a hard-won lesson:
    \begin{itemize}\small
        \item \textbf{Journalism} --- freedom-of-information laws and the muckraking tradition established that records, public \emph{and} private, are legitimate targets of scrutiny. (You work in this lineage when you file a records request.)
        \item \textbf{Law} --- discovery rules compel disclosure; due process demands decisions be explainable and reviewable --- inherited almost verbatim by algorithmic accountability.
        \item \textbf{Accounting} --- after 1929, mandatory audits turned private finances into public record. Lesson: trustworthy disclosure at scale needs \textbf{standards and independent examiners}, not goodwill.
        \item \textbf{Urban planning} --- impact statements and public comment periods make community input a precondition for decisions about shared resources.
        \item \textbf{Engineering} --- licensure and safety codes established duties to the public that override duties to an employer. Data science is still writing its equivalent.
    \end{itemize}
    \bottomcite{Textbook Ch.~3, ``Public Interest Lineages.'' Missing Manual Ch.~8: ``Artifacts Have Politics.''}
\end{frame}

\begin{frame}{Working inside it --- openness, oversight, ownership}
    The counter-values are not a fix for the post-API age; they are a \textbf{stance} for working inside it. They frame everything we do today.
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \begin{block}{Openness}
                \footnotesize Make your \textit{own} work resist enclosure: transparent pipelines, documented collection, and redundant deposits (Zenodo, Internet Archive, open licenses). Wikipedia is the standing counter-example.
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{Oversight}
                \footnotesize Insist consequential systems be inspectable by parties who do \textit{not} own them: audits, documentation standards, DSA Article~40. \textbf{The skills this course teaches are oversight skills.}
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{Ownership}
                \footnotesize Prefer community-governed infrastructure: federated protocols (ActivityPub, AT~Protocol), data cooperatives, ArchiveTeam. The long-horizon answer to enclosure.
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.4em}
    \bottomcite{Textbook Ch.~3, ``Three Counter-Values.'' Federated platforms return in Ch.~12 (Social).}
\end{frame}

\begin{frame}[fragile]{Openness in practice --- document the collection}
    The chapter has no scraping code --- but openness \textit{does} have a concrete artifact. Record not just \textbf{what} you collected but \textbf{how}: endpoints, parameters, dates, and the platform's constraints \parencite{gebru2021datasheets}. A future reader (or you, in a year) can then reconstruct exactly what was allowed:
\begin{lstlisting}
collection = {
    "source":      "reddit.com  (PRAW, r/ClimateChange)",
    "endpoint":    "GET /r/{sub}/top   params: t=year, limit=1000",
    "collected":   "2026-02-14",
    "access_tier": "researcher program (application-gated)",
    "constraints": "no redistribution of raw comments; 100 req/min",
    "known_gaps":  "pre-2023 history unavailable -- Pushshift offline",
}
import json, pathlib
pathlib.Path("datasheet.json").write_text(json.dumps(collection, indent=2))
# The datasheet travels next to the data; the source may change, the record does not.
\end{lstlisting}
    \bottomcite{Textbook Ch.~3, ``Openness'' --- documentation as the practice that survives enclosure.}
\end{frame}

\begin{frame}{Key takeaways}
    \begin{enumerate}
        \item The open-API era ($\sim$2008--2018) was a \textbf{strategic bargain}, not a natural state --- and computational social science was built on the trade.
        \item \textbf{Enclosure} converts collective data into private assets: shutdowns, repricing, and AI-training licensing deals.
        \item \textbf{Exemption} is the asymmetry of scrutiny --- platforms study users freely while blocking those who study platforms.
        \item \textbf{Erosion} is infrastructural decay, so reproducibility is a problem you must solve \textit{at collection time}.
        \item The response is \textbf{redundancy}: openness, oversight, ownership --- and mastery of many access techniques so no single closure ends your research.
    \end{enumerate}
