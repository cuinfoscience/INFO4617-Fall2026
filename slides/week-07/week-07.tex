\section{Monday · Concepts}
% =========================================================================

\begin{frame}{The web is not permanent}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            Pages change, disappear, and get redesigned constantly. A large fraction of URLs break within a few years --- \textbf{link rot}.

            \bigskip

            This is a research problem: how do you study something that might not be there tomorrow, or measure how it \textit{changed} when only today's version is visible?

            \bigskip

            The \textbf{Internet Archive} (Brewster Kahle, 1996) answers by archiving the web. Its \textbf{Wayback Machine} has captured over 800 billion pages, free to anyone.

            \bigskip

            It is both a data source and \textit{counter-erosion infrastructure} --- a community-governed public resource (the book's ``post-API'' ownership theme).
        \end{column}
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/wayback_calendar.png}\\
            {\scriptsize The Wayback calendar: every dot is a capture you can retrieve.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~7, ``The Web Is Not Permanent.''}
\end{frame}

\begin{frame}{From snapshots to a longitudinal source}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            One old snapshot is a curiosity. \textbf{Comparing snapshots across time} is where the research value lives.

            \bigskip

            The Wayback Machine turns the web from a snapshot medium into a \textbf{longitudinal data source} --- you can \textit{measure} change, not merely observe it:
            \begin{itemize}\small
                \item Track a platform's Terms of Service across a decade
                \item Follow a government page across administrations
                \item Contrast two rivals' homepages as they evolve
            \end{itemize}

            \bigskip

            You are not just reading old pages --- you are measuring \textbf{institutional behavior over time}.
        \end{column}
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/change_timeline.png}\\
            {\scriptsize A ``change timeline'' we build Wednesday: spikes mark major revisions.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~7, ``Comparing Multiple Snapshots'' \& ``Tracking Policy Changes.''}
\end{frame}

\begin{frame}{Raw or wrapped? The \texttt{id\_} rule}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            A snapshot URL looks like:\\
            {\scriptsize\texttt{web.archive.org/web/\textbf{20000815052826}/http://cnn.com/}}

            \bigskip

            By default the Archive \textbf{injects} its own toolbar, banner, and JavaScript, and \textbf{rewrites} every link, script, and image to point back into the archive. Convenient for browsing --- \textbf{poison for measurement}: your link, script, and image counts include the scaffolding.

            \bigskip

            Append \texttt{id\_} to the timestamp\\
            {\scriptsize\texttt{.../20000815052826\textbf{id\_}/http://cnn.com/}}\\
            to get the \textbf{original, unmodified capture}. Raw doesn't render prettily --- but for measurement, raw is exactly what you want.
        \end{column}
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/wrapped_vs_raw.png}\\
            {\scriptsize Wrapped (toolbar + rewritten links) vs.\ the \texttt{id\_} raw capture.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~7, ``The Availability API.'' See also \textcite{brugger_UnderstandingArchivedWeb_2019} on archived-web caveats.}
\end{frame}

\begin{frame}{What the notebook covers}
    This week's companion notebook builds one skill at a time:
    \begin{itemize}
        \item \textbf{Availability API} --- find the snapshot closest to a date; retrieve it raw with \texttt{id\_}
        \item \textbf{CDX Server API} --- enumerate a URL's \textit{entire} capture history, filtered server-side
        \item \textbf{Policy tracking} --- loop over quarterly dates, collect a document's history, serialize to JSON
        \item \textbf{Bag-of-words} --- tokenize, drop stopwords, and measure how much consecutive versions changed
    \end{itemize}

    \bigskip

\begin{frame}[fragile]{Finding a snapshot --- the Availability API}
    The simplest question: does a snapshot exist near a date? Pass query args as a \texttt{params} dict so \texttt{requests} URL-encodes them (one value is itself a URL):
\begin{lstlisting}
import requests

api_url = "https://archive.org/wayback/available"
response = requests.get(api_url, params={"url": "https://www.cnn.com",
                                         "timestamp": "20000815"})  # Aug 15, 2000
data = response.json()

if data["archived_snapshots"]:
    snap = data["archived_snapshots"]["closest"]   # closest capture to that date
    print(snap["timestamp"], snap["url"])
else:
    print("No snapshot available near this date")
\end{lstlisting}
    \bottomcite{Textbook Ch.~7, ``The Availability API.''}
\end{frame}

\begin{frame}[fragile]{Retrieve it raw with \texttt{id\_}, then measure}
    Rewrite the snapshot URL to the \texttt{id\_} form, then parse it like any page:
\begin{lstlisting}
from bs4 import BeautifulSoup

ts = snap["timestamp"]
raw_url = snap["url"].replace(ts, ts + "id_")   # id_ -> original, unwrapped capture

soup = BeautifulSoup(requests.get(raw_url).text, "html.parser")
stats = {"year": ts[:4], "num_links": len(soup.find_all("a")),
         "num_scripts": len(soup.find_all("script"))}
\end{lstlisting}
    The same measurement across three eras tells the web's story:
\begin{lstlisting}
#   year  num_links  text_length  num_images  num_scripts
#   2000        150         8500          20            2   # largely static HTML
#   2024        800        45000         120           95   # a JS app that renders
\end{lstlisting}
    \bottomcite{Textbook Ch.~7, ``Comparing Multiple Snapshots.'' Without \texttt{id\_}, every count is inflated by the toolbar.}
\end{frame}

\begin{frame}[fragile]{The whole history --- the CDX Server API}
    Availability finds \textit{one} snapshot; CDX enumerates \textit{all} of them. Two params filter server-side --- less data, less cleanup:
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
\begin{lstlisting}
cdx = "https://web.archive.org/cdx/search/cdx"
params = {"url": "facebook.com/terms",
          "output": "json",
          "fl": "timestamp,original,statuscode,length",
          "filter": "statuscode:200",  # only 200s
          "collapse": "digest"}        # drop repeats
data = requests.get(cdx, params=params).json()

df = pd.DataFrame(data[1:], columns=data[0])
df["timestamp"] = pd.to_datetime(
    df["timestamp"], format="%Y%m%d%H%M%S")
\end{lstlisting}
        \end{column}
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/cdx_response.png}\\
            {\scriptsize Row 0 is the header; each later row is one distinct capture.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~7, ``The CDX Server API.'' Add \texttt{"limit": 100} while developing.}
\end{frame}

\begin{frame}[fragile]{Tracking a document over time}
    Wrap the Availability + \texttt{id\_} + parse steps in \texttt{get\_wayback\_content()} (returns \texttt{None} on a miss), then loop over quarterly dates:
\begin{lstlisting}
dates = pd.date_range(start="2005-01-01", end="2024-01-01", freq="QS")

results = []
for date in dates:
    snap = get_wayback_content("facebook.com/terms", date)
    if snap is not None:          # skip misses instead of crashing the run
        results.append(snap)
    time.sleep(1)                 # rest between requests -- the Archive is a nonprofit
# Retrieved 68 snapshots from 77 time points
\end{lstlisting}
    Serialize immediately --- you may not be able to re-scrape it later:
\begin{lstlisting}
import json
with open("fb_tos_history.json", "w") as f:
    json.dump(results, f, indent=2)   # (also cache each raw capture as you go)
\end{lstlisting}
    \bottomcite{Textbook Ch.~7, ``Retrieving Historical Content,'' ``Serializing,'' \& ``Caching Raw Captures.''}
\end{frame}

\begin{frame}[fragile]{Measuring change --- bag-of-words}
    Represent each version as a vector of word frequencies (order ignored), then compare vectors:
\begin{lstlisting}
from gensim.utils import simple_preprocess
from gensim import corpora, similarities
from nltk.corpus import stopwords          # nltk.download("stopwords") once

stop = set(stopwords.words("english"))
docs = [[t for t in simple_preprocess(r["content"]) if t not in stop] for r in results]

dct = corpora.Dictionary(docs)             # word -> id
corpus = [dct.doc2bow(d) for d in docs]    # each version -> word-count vector
index = similarities.SparseMatrixSimilarity(corpus, num_features=len(dct))
change = [1 - index[corpus[i-1]][i] for i in range(1, len(corpus))]  # 0=same, 1=all-new
\end{lstlisting}
    Plotting \texttt{change} over time is the timeline from Monday: spikes flag major revisions --- a new policy, a redesign, a response to controversy.
    \bottomcite{Textbook Ch.~7, ``Document Similarity with Bag-of-Words'' \& ``Visualizing Document Change.''}
\end{frame}

\begin{frame}{Key takeaways}
    \begin{enumerate}
        \item The web is impermanent (link rot); the \textbf{Internet Archive} is public infrastructure that preserves it.
        \item \textbf{Availability API} finds one snapshot near a date; the \textbf{CDX Server} enumerates the full history --- filter and collapse server-side.
        \item Append \texttt{id\_} for the \textbf{raw} capture; the toolbar-wrapped version pollutes every measurement.
        \item Wayback $+$ text analysis turns the archive into \textbf{longitudinal data} --- measure change, don't just observe it.
        \item Be a good citizen: \texttt{sleep} between requests, cache raw captures, and \textbf{serialize} --- you may not be able to re-scrape.
    \end{enumerate}
