\section{Monday · Concepts}
% =========================================================================

\begin{frame}{When static scraping fails}
    \begin{columns}[T]
        \begin{column}{0.58\textwidth}
            Last week's tools --- \texttt{requests} + BeautifulSoup --- work on \textbf{static} pages, whose content is fully present in the initial HTML.

            \vspace{1em}

            But most modern sites are \textbf{dynamic}: their content is loaded or rendered by JavaScript \textit{after} the HTML arrives. \texttt{requests.get()} hands you the skeleton \textit{before} that code has run.

            \vspace{1em}

            The data you see in your browser may simply not exist in the response. Diagnose it: compare \texttt{requests.get(url).text} to the page in your browser --- if the content you want is missing, the page is dynamic.

            \vspace{1em}

            The fix: a tool that \textit{executes} JavaScript --- a real browser, driven by code. That tool is \textbf{Selenium}.
        \end{column}
        \begin{column}{0.38\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/requests_vs_browser.png}\\
            {\scriptsize Same URL, two very different payloads.}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~8, ``When Static Scraping Fails.''}
\end{frame}

\begin{frame}{Selenium: a browser you drive with code}
    \begin{columns}[T]
        \begin{column}{0.56\textwidth}
            Selenium needs two things: the \texttt{selenium} library and a \textbf{browser driver} --- a small program that lets Python control a specific browser.

            \vspace{1em}

            Since v4.6 (late 2022), \textbf{Selenium Manager} finds or downloads the right driver automatically, so in practice you only \texttt{pip install selenium}. Creating a driver is a one-liner:
            \begin{itemize}\small
                \item \texttt{driver = webdriver.Chrome()} opens a real window
                \item every command you issue through \texttt{driver} happens \textit{in that window}
            \end{itemize}

            \vspace{1em}

            You are not faking HTTP requests any more --- you are \textbf{automating a browser}, the same one a person would use.
        \end{column}
        \begin{column}{0.38\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/selenium_browser.png}\\
            {\scriptsize ``Controlled by automated test software.''}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~8, ``Setting Up Selenium.''}
\end{frame}

\begin{frame}{A powerful last resort --- and not a loophole}
    \begin{columns}[T]
        \begin{column}{0.56\textwidth}
            Selenium is slow, resource-intensive, and brittle. Use it only when the simpler tools cannot. The decision tree:
            \begin{enumerate}\small
                \item \textbf{API?} Use it --- fastest, most structured.
                \item \textbf{In the static HTML?} \texttt{requests} + BeautifulSoup.
                \item \textbf{Rendered by JavaScript?} \textit{Now} reach for a browser.
            \end{enumerate}

            \vspace{0.5em}

            \textbf{Playwright} (Microsoft) is the modern alternative --- auto-waiting, faster, async --- but Selenium is more widespread and fine for coursework.

            \vspace{0.5em}

            Researchers increasingly \textit{need} automation as platforms enclose once-open data \parencite{freelon2018computational} --- which raises the stakes, not lowers them.
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=0.82\textwidth]{img/decision_tree.png}\\[0.4em]
            \begin{alertblock}{Automation is not a loophole}
                \footnotesize A \texttt{robots.txt} rule and a site's ToS still apply when a real browser does the clicking. (Ch.~2)
            \end{alertblock}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~8, ``When to Use Selenium vs.\ APIs'' \& ``The Fragility Problem.''}
\end{frame}

    \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Launch a driver, then locate elements}
    Four strategies find elements; \texttt{find\_element} returns the first match, \texttt{find\_elements} returns a list:
\begin{lstlisting}
from selenium import webdriver
from selenium.webdriver.common.by import By

driver = webdriver.Chrome()          # Selenium Manager fetches the driver
driver.get("https://xkcd.com")

comic = driver.find_element(By.ID, "comic")                        # by id
nav   = driver.find_elements(By.CSS_SELECTOR, "ul.comicNav li a")  # -> a list
title = driver.find_element(By.XPATH, "//div[@id='ctitle']")       # by XPath
print(title.text)                    # the title of the current comic
\end{lstlisting}
    Prefer a stable anchor --- an \texttt{id} or a semantic \texttt{class} --- over a brittle, position-based path.
    \bottomcite{Textbook Ch.~8, ``Navigating and Finding Elements.''}
\end{frame}

\begin{frame}[fragile]{Read attributes, then act like a user}
\begin{lstlisting}
# xkcd hides a joke in the image's title (hover) attribute
img = driver.find_element(By.XPATH, "//div[@id='comic']//img")
print(img.get_attribute("alt"))      # the accessible description
print(img.get_attribute("title"))    # the mouseover punchline

# Anything a user can do, Selenium can do -- e.g. click "Random":
btn = driver.find_element(
    By.XPATH, "//ul[@class='comicNav']//a[contains(text(),'Random')]")
btn.click()
\end{lstlisting}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            \texttt{get\_attribute()} reads \textit{any} HTML attribute (even ones you never see); \texttt{.text} reads the visible text.
        \end{column}
        \begin{column}{0.36\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/xkcd_inspect.png}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~8, ``Extracting Data'' \& ``Simulating Interactions.''}
\end{frame}

\begin{frame}[fragile]{Type, submit --- then wait \textit{well}}
    Fill a search box, submit, and wait for the exact condition instead of guessing a duration:
\begin{lstlisting}
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

driver.get("https://www.wikipedia.org")
box = driver.find_element(By.NAME, "search")
box.send_keys("Colorado Buffaloes")      # type like a user
box.send_keys(Keys.RETURN)               # submit the form

WebDriverWait(driver, 10).until(         # not a blind time.sleep(2)!
    EC.presence_of_element_located((By.ID, "firstHeading")))
print(driver.title)                      # -> Colorado Buffaloes - Wikipedia
\end{lstlisting}
    \texttt{WebDriverWait} polls every 0.5\,s, returns \textit{the instant} the condition holds, and raises \texttt{TimeoutException} if it never does --- faster, and it fails loudly. Common conditions: \texttt{presence\_of\_element\_located}, \texttt{visibility\_of\_element\_located}, \texttt{element\_to\_be\_clickable}.
    \bottomcite{Textbook Ch.~8, ``Typing and Searching'' \& ``Waiting for Dynamic Content.''}
\end{frame}

\begin{frame}[fragile]{Scraping an infinite-scroll page}
    Scroll to the bottom, wait, and stop when the page stops growing:
\begin{lstlisting}
def scrape_infinite_scroll(driver, max_scrolls=20, pause=2):
    last = driver.execute_script("return document.body.scrollHeight")
    for i in range(max_scrolls):
        driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")
        time.sleep(pause)                # let new content load
        new = driver.execute_script("return document.body.scrollHeight")
        if new == last:                  # height stopped growing -> the end
            break
        last = new
    return driver.page_source            # the fully-loaded HTML
\end{lstlisting}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            Set a sane \texttt{max\_scrolls}. \textbf{Proportionality} (Ch.~2): collect only what your question needs --- not an entire 10,000-post feed.
        \end{column}
        \begin{column}{0.36\textwidth}
            \centering
            \includegraphics[width=\textwidth]{img/infinite_scroll.png}
        \end{column}
    \end{columns}
    \bottomcite{Textbook Ch.~8, ``Scraping Infinite Scroll Pages.''}
\end{frame}

\begin{frame}[fragile]{Selenium renders; BeautifulSoup reads}
    Get the page into the state you need, hand the rendered HTML to BeautifulSoup, and \textbf{always} clean up:
\begin{lstlisting}
from bs4 import BeautifulSoup

try:
    driver.get("https://example-search-site.com")
    # ...scroll / click / search until the page is ready...
    soup = BeautifulSoup(driver.page_source, "html.parser")  # rendered HTML
    for item in soup.find_all("div", class_="result-item"):
        title = item.find("h3")
        if title is None:            # same defensive habit as static scraping
            continue                 # skip a malformed item instead of crashing
        print(title.get_text(strip=True))
finally:
    driver.quit()                    # each driver is a whole browser process
\end{lstlisting}
    Selenium is the \textbf{hands} that navigate; BeautifulSoup is the \textbf{eyes} that read. \texttt{try/finally} guarantees the browser closes even when parsing throws.
    \bottomcite{Textbook Ch.~8, ``Passing to BeautifulSoup'' \& ``A Practical Selenium Workflow.''}
\end{frame}

\begin{frame}{Key takeaways}
    \begin{enumerate}
        \item Static tools see the HTML \textbf{skeleton}; dynamic pages render their data with JavaScript \textit{after} it arrives.
        \item Selenium drives a \textbf{real browser} --- locate elements (ID / name / CSS / XPath), then click, type, and scroll like a user.
        \item Wait on \textbf{conditions} with \texttt{WebDriverWait}, not a blind \texttt{time.sleep()} --- it's faster and fails loudly.
        \item Render with Selenium, extract with BeautifulSoup --- hands and eyes --- and \texttt{driver.quit()} every time.
        \item Selenium is a slow, fragile \textbf{last resort}: API~$\rightarrow$~static~$\rightarrow$~browser. Automation never suspends \texttt{robots.txt} or a site's ToS.
    \end{enumerate}
