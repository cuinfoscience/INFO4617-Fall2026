\begin{frame}{Public data by mandate}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            Government APIs exist because of democratic commitments to \textbf{transparency} --- not business strategy:
            \begin{itemize}\small
                \item The \textbf{Census Bureau} is constitutionally mandated to count the population.
                \item The \textbf{Federal Reserve} publishes the economic data it collects; FRED is the St.\ Louis Fed's service for it.
                \item The \textbf{Energy Information Administration} was created in 1977, after the energy crises, to publish independent energy statistics.
                \item The \textbf{Library of Congress} publishes Congress's legislative record at Congress.gov.
            \end{itemize}

            \medskip

            They embody the \textbf{openness} value from Ch.~3, \textit{The Post-API Age}: public data, freely accessible by mandate rather than platform goodwill.
        \end{column}
        \begin{column}{0.35\textwidth}
            \begin{alertblock}{Stable, but less polished}
                \footnotesize Agencies don't pivot business models or get acquired --- so these APIs are unusually \textbf{stable}. The tradeoff: dense docs, cryptic variable names, unhelpful errors. Navigating that \textit{is} the skill.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Keys stay out of the notebook}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            A \textbf{key} identifies you, counts your requests, and enforces rate limits.

            \medskip
            A key typed into a notebook travels with it: into a commit, a submission, a screenshot, a classmate's laptop.

            \medskip
            So treat it like a password. Never paste it into a cell or a code file, and never print it.
        \end{column}
        \begin{column}{0.35\textwidth}
            \textbf{Three safe ways in}\\[2pt]
            {\footnotesize An environment variable, set once (next frame). A JSON file kept out of git. \texttt{getpass()}, when you can't restart.\par}
            \medskip
            \textbf{Names}\\[2pt]
            {\footnotesize The lab uses \texttt{CENSUS\_KEY} and \texttt{FRED\_KEY}. Chapter 11 uses the same names, and \texttt{EIA\_KEY} for the EIA.\par}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{\texttt{KeyError}: Python can't see the key}
    \begin{columns}[T]
        \begin{column}{0.37\textwidth}
            Jupyter sees only the variables set \textbf{before it started}, in \textbf{the place it started from}.

            \medskip
            Chapter 11's ``When a Key Says \texttt{MISSING}'' lists the usual causes.
        \end{column}
        \begin{column}{0.6\textwidth}
            \begin{itemize}\footnotesize
                \item Jupyter was already running, or started from somewhere else: Navigator (which, on a Mac, doesn't read \texttt{\textasciitilde/.zshrc}), another terminal, VS Code
                \item \texttt{!export} in a notebook cell: its shell closes at once, and the key is now in the notebook
                \item Windows has no \texttt{export}; \texttt{setx} reaches only prompts opened after it
                \item \texttt{conda env config vars set}, without activating the environment again
                \item A different name: \texttt{CENSUS\_KEY}, not \texttt{census\_key} or \texttt{CENSUS\_API\_KEY}
            \end{itemize}
            \medskip
            {\footnotesize \textbf{Stuck at \texttt{[*]}?} \texttt{getpass()} is waiting: look for its box, paste, press Enter.\par}

\begin{frame}{Discovery: two catalogs before any data}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \dayband{What}{the variable catalog}
            \begin{itemize}\small
                \item Variables come in \textbf{groups}: the website's \textit{table} \texttt{B19013} is the API's \textit{group}
                \item \texttt{B19013\_001E}: group, line, and \texttt{E} for estimate (\texttt{M} is its margin of error)

            \begin{itemize}\small
                \item \texttt{.../acs5/geography.json} lists every level the dataset answers for
                \item Each says what it \textbf{requires}: a \texttt{place} needs a \texttt{state}; a \texttt{tract} needs a \texttt{state} and a \texttt{county}

            \end{itemize}
        \end{column}
    \end{columns}

    \bigskip
    \centering\small Codes change between survey years: learn to look them up, not to memorize them. Then check one value against the website.

\begin{frame}[fragile]{Step 1: the search, given}
    \begin{columns}[T]

            A given cell names the search: a \textbf{function}. It runs on whatever words you give it.

# Given: the search, for any words
def find_groups(words):
    match = groups["description"].str.contains(
        words,
        case=False,
        regex=False,
    )
    return groups[match].sort_values("name")

\begin{frame}{Key takeaways}
    \begin{enumerate}
        \item Government APIs serve \textbf{public mandates}: the census, the Federal Reserve's data.
        \item Each carried one idea you'll meet again: \textbf{ask what you can ask for} (Census), \textbf{let the server compute} (FRED).
        \item \textbf{Start from the website, and start small}: one place, one year, or three months you can check, before the whole table.
        \item \textbf{Read what isn't a number}: a margin of error, a code like \texttt{-666666666}, a cap at \$3,500, a \texttt{"."} for missing.
        \item \textbf{Treat keys like passwords}: an environment variable, a JSON file kept out of git, or \texttt{getpass()}; never in a cell or a code file, and never printed.
    \end{enumerate}
\end{frame}
